```sql
WITH batch(t) AS (VALUES ('ai_generation_logs'),('bank_transactions_enrichment_audit'),('channel_price_lists'),('channel_pricing_history'),('channel_product_metadata'),('channel_stats_snapshots'),('client_consultations'),('collection_shares'),('consultation_emails'),('consultation_images'),('consultation_products'),('customer_groups'),('customer_pricing'),('document_emails'),('email_templates'),('feed_configs'),('finance_settings'),('fiscal_obligations_done'),('fixed_asset_depreciations'),('fixed_assets'),('gmail_watch_state'),('google_merchant_syncs'),('group_price_lists'),('linkme_channel_suppliers'),('matching_rules'),('mcp_resolution_queue'),('mcp_resolution_strategies'),('media_asset_analytics'),('media_asset_publications'),('meta_commerce_syncs'),('order_payments'),('organisation_families'),('pinterest_pin_syncs'),('price_list_history'),('price_list_items'),('price_lists'),('product_colors'),('product_commission_history'),('product_packages'),('product_purchase_history'),('purchase_order_receptions'),('sample_order_items'),('sample_orders'),('scheduled_publications'),('sourcing_candidate_suppliers'),('sourcing_communications'),('sourcing_photos'),('sourcing_price_history'),('sourcing_urls'),('stock_reservations'),('storage_allocations'),('sync_runs'),('transaction_document_links'),('user_notification_preferences'),('webhook_configs'),('webhook_logs')),
anon_tables AS (
  SELECT c.oid, c.relname, c.relkind, c.relrowsecurity
  FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
  WHERE n.nspname='public' AND c.relkind IN ('r','p','v','m')
    AND has_table_privilege('anon', c.oid, 'SELECT')
)
SELECT a.relname, a.relkind, a.relrowsecurity,
  (a.relname IN (SELECT t FROM batch)) AS in_batch,
  EXISTS (SELECT 1 FROM pg_policies p WHERE p.schemaname='public' AND p.tablename=a.relname AND (p.roles && ARRAY['anon','public']::name[])) AS has_anon_policy,
  (SELECT string_agg(p.policyname||'['||p.cmd||':'||array_to_string(p.roles,',')||']', ' ; ') FROM pg_policies p WHERE p.schemaname='public' AND p.tablename=a.relname AND (p.roles && ARRAY['anon','public']::name[])) AS anon_policies,
  has_table_privilege('anon', a.oid, 'INSERT') AS ins, has_table_privilege('anon', a.oid, 'UPDATE') AS upd, has_table_privilege('anon', a.oid, 'DELETE') AS del
FROM anon_tables a
ORDER BY a.relkind, has_anon_policy, in_batch, a.relname;
```

### A2. Appels `anon` par table (total + requête la plus fréquente)

```sql
WITH t(name) AS (VALUES ('affiliate_archive_requests'),('affiliate_storage_requests'),('ambassador_attributions'),('ambassador_codes'),('audit_logs'),('bank_transactions'),('brands'),('channel_pricing'),('collection_images'),('contacts'),('counterparty_bank_accounts'),('customer_addresses'),('email_messages'),('families'),('financial_document_items'),('financial_documents'),('individual_customers'),('linkme_commissions'),('linkme_info_requests'),('linkme_onboarding_progress'),('linkme_page_configurations'),('linkme_payment_request_items'),('linkme_payment_requests'),('linkme_payments'),('media_assets'),('newsletter_subscribers'),('order_discount_targets'),('order_discounts'),('promotion_usages'),('sales_channels'),('sales_order_items'),('sales_order_linkme_details'),('sales_order_shipments'),('sales_orders'),('site_contact_messages'),('stock_alert_tracking'),('storage_pricing_tiers'),('user_profiles'),('variant_groups'),('wishlist_items')),
anon_q AS (
  SELECT s.queryid, s.calls, s.query
  FROM pg_stat_statements s JOIN pg_roles r ON r.oid = s.userid
  WHERE r.rolname = 'anon'
)
SELECT t.name,
  COALESCE(SUM(q.calls),0) AS anon_calls,
  COUNT(q.queryid) AS distinct_queries,
  LEFT(regexp_replace((array_agg(q.query ORDER BY q.calls DESC))[1], '\s+', ' ', 'g'), 160) AS top_query
FROM t
LEFT JOIN anon_q q ON (q.query ILIKE '%public.' || t.name || '%' OR q.query ILIKE '%"' || t.name || '"%' OR q.query ILIKE '% ' || t.name || ' %')
GROUP BY t.name
ORDER BY anon_calls DESC, t.name;
```

```sql
FROM t JOIN anon_q q ON (q.query ILIKE '%public.' || t.name || '%' OR q.query ILIKE '%"' || t.name || '"%' OR q.query ILIKE '% ' || t.name || ' %')
ORDER BY t.name, q.calls DESC;
```

### A5. Toutes les policies des tables à lecteur public

```sql
SELECT tablename, policyname, cmd, array_to_string(roles, ',') AS roles, LEFT(regexp_replace(COALESCE(qual, with_check, ''), '\s+', ' ', 'g'), 110) AS expr
FROM pg_policies
WHERE schemaname='public' AND tablename IN ('newsletter_subscribers','wishlist_items','linkme_page_configurations','channel_pricing','order_discounts','order_discount_targets','promotion_usages','collection_images','sales_order_linkme_details','sales_order_items','sales_orders','customer_addresses','site_contact_messages')
ORDER BY tablename, policyname;
```
